    <script>
        function openModal(id) { document.getElementById(id).style.display = "flex"; }
        function closeModal(id) { document.getElementById(id).style.display = "none"; }
        window.onclick = function(event) { if (event.target.classList.contains('modal')) event.target.style.display = "none"; }

        function contactAdmin() { window.open('https://wa.me/6285795941412', '_blank'); }

        // PASTE URL GAS PENDAFTARAN LAMA ANDA DI SINI
        const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbyMXZW_hSs8F84f2wDWBcUxKcgbecaToyT2OgJ7kd1faohHk56lbytoh764omSQPXItUQ/exec'; 

        document.getElementById('loginForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const name = document.getElementById('loginName').value.trim();
            const wa = document.getElementById('loginWa').value.trim();
            const btn = document.getElementById('btnLogin');
            const msg = document.getElementById('loginMessage');
            
            if(!name || !wa) {
                msg.innerText = "Mohon isi Nama Lengkap dan Nomor WhatsApp terlebih dahulu!";
                return;
            }

            btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Memeriksa Database...'; 
            btn.disabled = true; msg.innerText = "";
            document.getElementById('confirmSection').classList.add('hidden');
            document.getElementById('loginSection').classList.remove('hidden'); 

            localStorage.setItem('userName', name);
            localStorage.setItem('userWa', wa);

            try {
                const url = `${SCRIPT_URL}?action=login&name=${encodeURIComponent(name)}&wa=${encodeURIComponent(wa)}`;
                const response = await fetch(url, { redirect: 'follow' });
                
                if(!response.ok) throw new Error("Network response was not ok");
                const result = await response.json();

                if(result.status === 'success') { 
                    window.location.href = 'dashboard.html'; 
                } 
                else if (result.status === 'name_match_wa_mismatch') {
                    document.getElementById('confirmText').innerText = `Apakah Anda "${result.name}" yang bertugas di "${result.school}"? Jika Ya, Nomor WA yang Anda input salah.`;
                    document.getElementById('loginSection').classList.add('hidden');
                    document.getElementById('confirmSection').classList.remove('hidden');
                } 
                else if (result.status === 'wa_match_name_mismatch') {
                    msg.innerHTML = "⚠️ Nomor WA Anda terdata dengan Nama yang berbeda. Mohon ketik nama persis saat registrasi, atau <a href='javascript:void(0)' onclick='contactAdmin()' style='color:#b91c1c; text-decoration:underline;'>Hubungi Admin</a>.";
                    btn.innerText = "Masuk ke LMS"; btn.disabled = false;
                } 
                else { 
                    window.location.href = 'register.html'; 
                }
            } catch (error) {
                console.error(error);
                msg.innerText = "Gagal terhubung ke Database. Pastikan Anda memiliki koneksi internet.";
                btn.innerText = "Masuk ke LMS"; btn.disabled = false;
            }
        });
    </script>
